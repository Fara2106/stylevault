func load(_ d: Data) -> CGImage? {
    guard let s = CGImageSourceCreateWithData(d as CFData, nil) else { return nil }
    return CGImageSourceCreateImageAtIndex(s, 0, nil)
}
func save(_ img: CGImage, _ name: String) {
    let out = NSMutableData()
    let dest = CGImageDestinationCreateWithData(out, UTType.jpeg.identifier as CFString, 1, nil)!
    CGImageDestinationAddImage(dest, img, [kCGImageDestinationLossyCompressionQuality: 0.85] as CFDictionary)
    CGImageDestinationFinalize(dest)
    try? (out as Data).write(to: outDir.appendingPathComponent(name))
}
func panel(_ images: [CGImage], magenta: Bool = false) -> CGImage? {
    let H = 700.0
    let widths = images.map { Double($0.width) * H / Double($0.height) }
    let W = widths.reduce(0, +) + Double(images.count - 1) * 10
    guard let ctx = CGContext(data: nil, width: Int(W), height: Int(H), bitsPerComponent: 8, bytesPerRow: 0,
                              space: CGColorSpaceCreateDeviceRGB(), bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return nil }
    ctx.setFillColor(magenta ? CGColor(red: 1, green: 0.2, blue: 1, alpha: 1) : CGColor(gray: 1, alpha: 1)); ctx.fill(CGRect(x: 0, y: 0, width: W, height: H))
    var x = 0.0
    for (i, img) in images.enumerated() { ctx.draw(img, in: CGRect(x: x, y: 0, width: widths[i], height: H)); x += widths[i] + 10 }
    return ctx.makeImage()
}

/// Il primo riquadro di un foglio di confronto già salvato: la foto del corpo di allora (640 px di altezza).
func leftPanel(_ path: String) -> Data? {
    guard let d = try? Data(contentsOf: URL(fileURLWithPath: path)), let img = load(d) else { return nil }
    let w = img.width, h = img.height
    var px = [UInt8](repeating: 0, count: w * 4)
    let ctx = CGContext(data: &px, width: w, height: 1, bitsPerComponent: 8, bytesPerRow: w * 4, space: CGColorSpaceCreateDeviceRGB(), bitmapInfo: CGImageAlphaInfo.noneSkipLast.rawValue)!
    ctx.draw(img, in: CGRect(x: 0, y: -(h / 2), width: w, height: h))
    var run = 0, cut = 0
    for x in 50..<w {
        let white = px[x * 4] > 235 && px[x * 4 + 1] > 235 && px[x * 4 + 2] > 235
        run = white ? run + 1 : 0
        if run >= 8 { cut = x - run; break }
    }
    guard cut > 0, let crop = img.cropping(to: CGRect(x: 0, y: 0, width: cut, height: h)) else { return nil }
    let out = NSMutableData()
    let dest = CGImageDestinationCreateWithData(out, UTType.jpeg.identifier as CFString, 1, nil)!
    CGImageDestinationAddImage(dest, crop, [kCGImageDestinationLossyCompressionQuality: 0.95] as CFDictionary)
    CGImageDestinationFinalize(dest)
    return out as Data
}
